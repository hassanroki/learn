----------------------

TypeScript to JavaScript Compile:
---------------------------------
TypeScript Compiled JS Command: tsc filename.ts


Another way to compile:
------------------------
Create example.ts and tsconfig.json file

example.ts:
------------
function add(a: number, b: number): number {
    return a + b;
  }
  
  function subtract(a: number, b: number): number {
    return a - b;
  }
  
  function multiply(a: number, b: number): number {
    return a * b;
  }
  
  function divide(a: number, b: number): number {
    if (b === 0) {
      throw new Error("Cannot divide by zero");
    }
    return a / b;
  }
  
  console.log("add(2, 3) =", add(2, 3));
  console.log("subtract(5, 2) =", subtract(5, 2));
  console.log("multiply(4, 3) =", multiply(4, 3));
  console.log("divide(10, 2) =", divide(10, 2));

Intersection TypeScript:
-----------------------
type NameType = {
    firstName: string,
    lastName: string,
}

type ageType = {
    age: number
}

type DetailsType = NameType & ageType

function sayDetails(obj:DetailsType):string{
    let {firstName, lastName, age} = obj
    return `Full Name: ${firstName} ${lastName}, age: ${age}`;
}


let instructorDetails:DetailsType = {
    firstName: "Hasin",
    lastName: "Hyder",
    age: 10
}

console.log(sayDetails(instructorDetails))

function sayDetails(name:string, age:string | number):void{
    let currentAge:number;

    if(typeof age === "string"){
        currentAge = Number(age) - 2;
    }else{
        currentAge = age - 2;

    console.log(`Your name is: ${name}, your age is: ${currentAge}`)
}

11. Class in TypeScript:
------------------------
class Computer {
    private _ssd: number = 512;

    constructor(public name: string, public ram: number) {
        this.name = name;
        this.ram = ram;
    }

    // Private method
    private getDetails(): void {
        console.log(`Computer Name: ${this.name}`);
    }

    // get
    get ssdDetails(): number {
        return this._ssd;
    }
}

const computer1 = new Computer("HP", 16);





12. How to write an interface in a TypeScript class:
-----------------------------------------------------
interface ComputerInterface {
    name: string,
    ram: number
}

interface DetailsInterface {
    aboutComputer(): void;
}

class Computer implements ComputerInterface, DetailsInterface {
    constructor(public name: string, public ram: number) {
    }
    
    aboutComputer(): void {
        console.log(`Computer name: ${this.name}, and ram: ${this.ram}`);
    }
}

let computer10 = new Computer("Dell", 16)




13. Generics in TypeScript:
---------------------------
Example Normal Function:
-------------------------
function age(something:string):string{
    return something;
}

function age1(something:number):number{
    return something;
}

function age2(something:boolean):boolean{
    return something;
}


age("Ten");
age1(10);
age2(false);



Example use Generics Style:
---------------------------
function age<DynamicType>(something:DynamicType):DynamicType{
    return something;
}

age<string>("Ten");
age<number>(10);
age<boolean>(true);


Another Example Generics Style:
-------------------------------
function combineObj<T, U>(obj1:T, obj2:U){
    return {...obj1, ...obj2}
}

combineObj({name: "Hassan", age: 10}, {channel: 'Code ABC'});


Another example: must pass an object:
-------------------------------------
function combineObj<T extends object, U extends object>(obj1:T, obj2:U){
    return {...obj1, ...obj2}
}

// Wrong: must be an object passed
// combineObj({name: "Hassan", age: 10}, 'Code ABC');
combineObj({name: "Hassan", age: 10}, {channel: 'Code ABC'});

-----------------------

Set & Map Example:
------------------
// Restaurant Order System
const orders = new Map<string, number>()

function addOrder(juice: string){
    const quantity = (orders.get(juice) ?? 0) + 1
    orders.set(juice, quantity)
}

addOrder("lemon")
addOrder("lemon")
addOrder("apple")
addOrder("mango")
addOrder("mango")
addOrder("lemon")
